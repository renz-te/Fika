6. Payroll Processing

flowchart TD
    GEN["HR Clicks\nGenerate Payroll"] --> PERIOD["Select Period\nBi-monthly Cutoffs"]
    PERIOD --> FT{Employee Type?}
    FT -->|"Full-Time"| FT_CALC["Base Pay = Monthly Salary / 2\nDaily Rate = Salary / 22\nCount actual days attended + leaves\nDeduct unworked days"]
    FT -->|"Part-Time"| PT_CALC["Total Hours from Attendance\nx Hourly Rate"]
    FT_CALC --> DEDUCTIONS
    PT_CALC --> DEDUCTIONS
    DEDUCTIONS["Statutory Deductions\nPhilHealth 2.5% + SSS 4.5%\n+ Pag-IBIG P100 + Tax"]
    DEDUCTIONS --> BONUS["+ Pending Bonuses"]
    BONUS --> NET["Net Pay = Gross - Deductions + Bonus"]
    NET --> DRAFT["Status: Draft"]
    DRAFT -->|"HR Reviews"| APPROVE["Status: Approved"]
    APPROVE -->|"HR Releases"| RELEASE["Status: Released\n+ Payment Method\n+ Email Notification"]
